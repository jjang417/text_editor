// Added for HOST msix interrupt which is used in chiplib_f and set
// in the c11_dummy of host
bool sim_get_msix_intr(msix_intr_t &msix_intr)
{
    if (msixIntrQueue.size() == 0)
        return false;
    msix_intr = msixIntrQueue.front();
    msixIntrQueue.pop_front();
    return true;
}
