
  Summary: PCIe and C2C Transactions Between Vera and Remote CommLib GPU

  This document outlines the architectural integration paths for two distinct transaction types in a GPU simulation environment:

  PCIe Transaction Architecture

  The PCIe pathway comprises four binding layers:

  1. SoC-to-GPU Wrapper: Interrupt signals and memory buses connect the system die to the GPU wrapper through multiple PCIe interfaces (configuration, memory, VDM, display).
  2. GPU Wrapper-to-CommLib: The wrapper binds interrupt handlers and memory buses to the commlib GPU module, establishing bidirectional communication paths.
  3. CommLib-to-External Adapter: Four API categories handle external communication:
    - Configuration: Read/write operations targeting PCI configuration space via pcie_config_read/write
    - MMIO: Memory-mapped I/O access through pcie_mmio_read/write
    - System Memory: Direct system access callbacks for GPU memory operations
    - Legacy IRQ: Interrupt assertion/deassertion notifications
  4. Adapter-to-Remote GPU: The external adapter interfaces with the remote GPU's functional model through fmodel mutex-protected operations on configuration registers and bus memory
   blocks.

  C2C Transaction Architecture

  Cache-to-Cache transactions follow a parallel but distinct path:

  1. SoC-to-Wrapper Binding: Compute die connects to GPU wrapper via CHI protocol sockets.
  2. Wrapper-to-Transactor: A dedicated CHI transactor mediates between the wrapper and external commlib adapter.
  3. Transactor-to-Adapter: Request, data, and response packets flow through registered callbacks that decode request types (read, write, ATS, snoop).
  4. Adapter-to-Remote GPU: The commlib adapter establishes bidirectional callbacks with the remote GPU devproc module for processing memory requests, data transfers, and responses.

  Both pathways ultimately converge at the remote GPU's functional model through shared synchronization primitives.

    commLib_error_t ProcessMMIORead(PCIE_payload_t *payload)
    {
        TRACE_START(PCIE, "ProcessMMIORead start");

        LOG(PCIE, INFO, "MMIO_READ START: 0x%lx/%ld\n", payload->cpu_access.offset, payload->cpu_access.width);

        BusMemRet ret = BUSMEM_HANDLED;

    static BusMemRet BusMemBlkAccessDispatch(device_devproc_state_t *dsp, bool read, NvU064 address, void *appdata,
                                         NvU032 count)
    {
        PCIE_payload_t payload;
        PCIEConnection *connection = static_cast<PCIEConnection *>(dsp->pci_object);
        ;

        payload.dev_access.dfid = dsp->access_gfid;
        payload.dev_access.address = (uint64_t)address;
        payload.dev_access.count = (uint32_t)count;
        payload.dev_access.pcie_op = (read) ? pcie_op_read : pcie_op_write;
        //    payload.dev_access.device_inst = dsp->device_inst;

        if (read)
        {
            LOG(PCIE, INFO, "SYSMEM_READ START: 0x%lx/%d", payload.dev_access.address, payload.dev_access.count);
            if (connection->PCIESysmemRead(&payload))
            {
                LOG(PCIE, ERROR, "PCIESysmemRead call failed with following return value");
                return BUSMEM_NOTHANDLED;
            }
            std::memcpy(appdata, (void *)payload.dev_access.data, count);
            LOG(PCIE, INFO, "SYSMEM_READ DONE: 0x%lx/%d: 0x%lx", payload.dev_access.address, payload.dev_access.count,
                *(uint64_t *)&payload.dev_access.data);
        }
        else
        {
            LOG(PCIE, INFO, "SYSMEM_WRITE START: 0x%lx/%d", payload.dev_access.address, payload.dev_access.count);
            std::memcpy((void *)payload.dev_access.data, appdata, count);
            if (connection->PCIESysmemWrite(&payload))
            {
                LOG(PCIE, ERROR, "PCIESysmemWrite call failed with following return value");
                return BUSMEM_NOTHANDLED;
            }
            LOG(PCIE, INFO, "SYSMEM_WRITE DONE: 0x%lx/%d: 0x%lx", payload.dev_access.address, payload.dev_access.count,
                *(uint64_t *)&payload.dev_access.data);
        }

        return BUSMEM_HANDLED;
    }

    commLib_error_t ProcessAssertInterrupt(PCIE_payload_t *payload)
    {
        struct commlib_dev_info *commlib_info = (struct commlib_dev_info *)payload->interrupt.vmm_pcie_dev;
        uint32_t commlib_instance = commlib_info->instance_num;

        DEBUG("commlib instnace is " << commlib_instance << " GPU interrupt_id is " << payload->interrupt.interrupt_id);
        if (payload->interrupt.interrupt_id >= sizeof(uint32_t) * 8)
        {
            std::cout << "[ERROR] invalid gpu irq id" << std::endl;
            return commLib_failure;
        }

        if (!gpu_irq_assert_callback)
        {
            std::cout << "[ERROR] gpu_irq_assert_callback is NULL" << std::endl;
            return commLib_failure;
        }

        g_gpu_pending_irq_mask[commlib_instance] |= (0x1 << payload->interrupt.interrupt_id);

        // To nofify CPU there is a GPU IRQ assert if there is pending IRQ
        if (g_gpu_pending_irq_mask[commlib_instance])
        {
            gpu_irq_assert_callback(commlib_instance);
        }

        return commLib_success;
    }

    commLib_error_t ProcessDeassertInterrupt(PCIE_payload_t *payload)
    {
        struct commlib_dev_info *commlib_info = (struct commlib_dev_info *)payload->interrupt.vmm_pcie_dev;
        uint32_t commlib_instance = commlib_info->instance_num;

        DEBUG("commlib instnace is " << commlib_instance << " GPU interrupt_id is " << payload->interrupt.interrupt_id);
        if (payload->interrupt.interrupt_id >= sizeof(uint32_t) * 8)
        {
            std::cout << "[ERROR] invalid gpu irq id" << std::endl;
            return commLib_failure;
        }

        if (!gpu_irq_deassert_callback)
        {
            std::cout << "[ERROR] gpu_irq_deassert_callback is NULL" << std::endl;
            return commLib_failure;
        }

        g_gpu_pending_irq_mask[commlib_instance] &= ~(0x1 << payload->interrupt.interrupt_id);

        // To nofify CPU to deassert GPU IRQ if there is no pending IRQ
        if (g_gpu_pending_irq_mask[commlib_instance] == 0)

        return commLib_success;
    }
